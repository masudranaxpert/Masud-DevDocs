## Recursion কী?

Recursion হলো function যেটা নিজেকে নিজে call করে। প্রতিটা recursive solution এ দুটো অংশ থাকে:
1. **Base case** — recursion যেখানে থামে
2. **Recursive case** — নিজেকে নিজে ছোট problem দিয়ে call করে

```python
def countdown(n):
    if n <= 0:        # Base case
        print("Go!")
        return
    print(n)
    countdown(n - 1)  # Recursive case

countdown(5)   # → 5, 4, 3, 2, 1, Go!
```

## Call Stack

```python
def factorial(n):
    if n <= 1:
        return 1
    return n * factorial(n - 1)

factorial(5)
```

```
factorial(5)
  └─ 5 * factorial(4)
         └─ 4 * factorial(3)
                └─ 3 * factorial(2)
                       └─ 2 * factorial(1)
                              └─ return 1
                       return 2 * 1 = 2
                return 3 * 2 = 6
         return 4 * 6 = 24
  return 5 * 24 = 120
```

<Callout type="warn" title="RecursionError — Stack Overflow">
  Python এর default limit **1000**। Deep recursion এ `RecursionError` হয়।
  ```python
  import sys
  sys.getrecursionlimit()       # → 1000
  sys.setrecursionlimit(5000)   # সাবধানে বাড়ানো যায়
  ```
  Deep recursion → iterative solution বেশি safe।
</Callout>

## Classic Examples

### Fibonacci + Memoization

```python
# Naive — O(2^n), খুব SLOW
def fib_naive(n):
    if n <= 1:
        return n
    return fib_naive(n-1) + fib_naive(n-2)

# Memoized — O(n), FAST
from functools import lru_cache

@lru_cache(maxsize=None)
def fib(n):
    if n <= 1:
        return n
    return fib(n-1) + fib(n-2)

fib(100)    # → 354224848179261915075 (instant!)

# Manual memoization
def fib_memo(n, memo={}):
    if n in memo:
        return memo[n]
    if n <= 1:
        return n
    memo[n] = fib_memo(n-1, memo) + fib_memo(n-2, memo)
    return memo[n]
```

### Binary Search

```python
def binary_search(arr, target, low=0, high=None):
    if high is None:
        high = len(arr) - 1
    if low > high:
        return -1                  # Base case: not found
    mid = (low + high) // 2
    if arr[mid] == target:
        return mid                 # Base case: found
    elif arr[mid] < target:
        return binary_search(arr, target, mid+1, high)
    else:
        return binary_search(arr, target, low, mid-1)

arr = [1, 3, 5, 7, 9, 11]
binary_search(arr, 7)   # → 3
binary_search(arr, 4)   # → -1
```

### Fast Power — O(log n)

```python
def power(base, exp):
    if exp == 0:
        return 1
    if exp % 2 == 0:
        half = power(base, exp // 2)
        return half * half    # প্রতিবার problem অর্ধেক হয়
    return base * power(base, exp - 1)

# 2^10 = (2^5)^2 = (2*(2^4))^2 = (2*(2^2)^2)^2
power(2, 10)   # → 1024
```

## Tree Traversal — Recursion এর সেরা use case

```python
class TreeNode:
    def __init__(self, val, left=None, right=None):
        self.val, self.left, self.right = val, left, right

#       1
#      / \
#     2   3
#    / \
#   4   5
root = TreeNode(1, TreeNode(2, TreeNode(4), TreeNode(5)), TreeNode(3))

def inorder(node):    # left → root → right
    if node is None: return []
    return inorder(node.left) + [node.val] + inorder(node.right)

def height(node):
    if node is None: return 0
    return 1 + max(height(node.left), height(node.right))

def tree_sum(node):
    if node is None: return 0
    return node.val + tree_sum(node.left) + tree_sum(node.right)

inorder(root)   # → [4, 2, 5, 1, 3]
height(root)    # → 3
tree_sum(root)  # → 15
```

## Recursive Data Structures

```python
# Nested list flatten
def flatten(data):
    result = []
    for item in data:
        if isinstance(item, list):
            result.extend(flatten(item))
        else:
            result.append(item)
    return result

flatten([1, [2, [3, [4]], 5], 6])   # → [1, 2, 3, 4, 5, 6]

# Nested dict values
def all_values(d):
    for v in d.values():
        if isinstance(v, dict):
            yield from all_values(v)
        else:
            yield v

data = {"a": 1, "b": {"c": 2, "d": {"e": 3}}}
list(all_values(data))   # → [1, 2, 3]
```

## Backtracking

```python
# Maze solver — backtracking
def solve_maze(maze, row=0, col=0, path=None):
    if path is None:
        path = []

    rows, cols = len(maze), len(maze[0])

    # Base cases
    if row < 0 or row >= rows or col < 0 or col >= cols:
        return None
    if maze[row][col] == 1:   # wall
        return None
    if (row, col) in path:    # visited
        return None
    if row == rows-1 and col == cols-1:   # goal
        return path + [(row, col)]

    path = path + [(row, col)]

    # Try all 4 directions
    for dr, dc in [(1,0), (-1,0), (0,1), (0,-1)]:
        result = solve_maze(maze, row+dr, col+dc, path)
        if result:
            return result

    return None   # backtrack

maze = [
    [0, 0, 1, 0],
    [1, 0, 1, 0],
    [0, 0, 0, 0],
    [0, 1, 1, 0],
]
print(solve_maze(maze))
```

## Coin Change — Dynamic Programming

```python
from functools import lru_cache

@lru_cache(maxsize=None)
def min_coins(amount, coins=(1, 5, 10, 25)):
    if amount == 0: return 0
    if amount < 0:  return float('inf')
    return 1 + min(min_coins(amount - c, coins) for c in coins)

min_coins(30)   # → 2  (25+5)
min_coins(11)   # → 2  (10+1)
min_coins(67)   # → 5  (25+25+10+5+2? → 25+25+10+5+1+1 = 6? no, 25+25+10+5+1+1 = 67)
```

def reverse_string(s):

def is_palindrome(s):
    s = "".join(c.lower() for c in s if c.isalnum())
    if len(s) <= 1: return True
    if s[0] != s[-1]: return False
    return is_palindrome(s[1:-1])

# P3
def merge_sort(arr):
    if len(arr) <= 1: return arr
    mid = len(arr) // 2
    left, right = merge_sort(arr[:mid]), merge_sort(arr[mid:])
    result, i, j = [], 0, 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i]); i += 1
        else:
            result.append(right[j]); j += 1
    return result + left[i:] + right[j:]

# P4
def gcd(a, b):
    return a if b == 0 else gcd(b, a % b)

gcd(48, 18)   # → 6

# P5
def nested_sum(lst):
    total = 0
    for item in lst:
        if isinstance(item, list):
            total += nested_sum(item)
        else:
            total += item
    return total

# P6
def power_set(s):
    if not s: return [[]]
    rest = power_set(s[1:])
    return rest + [[s[0]] + sub for sub in rest]

power_set([1, 2, 3])
# → [[], [3], [2], [2,3], [1], [1,3], [1,2], [1,2,3]]

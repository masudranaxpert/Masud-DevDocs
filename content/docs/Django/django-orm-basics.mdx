## QuerySet Lazy Loading

```python
# Database hit নেই এখনো
posts = Post.objects.filter(status='published').order_by('-created_at')

# এখন database hit হয়:
for post in posts:    # iterate করলে
list(posts)           # list() করলে
posts[0]              # index access
posts[:5]             # slice
len(posts)            # len()
bool(posts)           # if posts:
```

## Filter Lookups

```python
# Text
Post.objects.filter(title__exact='Hello')
Post.objects.filter(title__iexact='hello')         # case-insensitive
Post.objects.filter(title__contains='Django')
Post.objects.filter(title__icontains='django')     # case-insensitive
Post.objects.filter(title__startswith='How')
Post.objects.filter(title__endswith='guide')

# Number
Post.objects.filter(salary__gt=50000)              # >
Post.objects.filter(salary__gte=50000)             # >=
Post.objects.filter(salary__lt=80000)              # <
Post.objects.filter(salary__lte=80000)             # <=
Post.objects.filter(salary__range=(40000, 80000))  # BETWEEN

# List / NULL
Post.objects.filter(category__in=['tech', 'news'])
Post.objects.filter(deleted_at__isnull=True)

# Date
Post.objects.filter(created_at__year=2024)
Post.objects.filter(created_at__month=3)
Post.objects.filter(created_at__date='2024-03-15')

# Related field — __ দিয়ে traverse
Post.objects.filter(author__username='rahim')
Post.objects.filter(author__profile__city='Dhaka')
```

## Annotate — Database Level Calculation

```python
from django.db.models import Count, Avg, Sum, Max, Min, F

# ❌ ভুল — N+1
for author in Author.objects.all():
    print(author.posts.count())  # প্রতিটার জন্য query!

# ✅ সঠিক — একটাই query
authors = Author.objects.annotate(post_count=Count('posts'))
for author in authors:
    print(author.post_count)  # no extra query

# Multiple annotations
Product.objects.annotate(
    avg_rating=Avg('reviews__rating'),
    review_count=Count('reviews'),
)

# F expression — field দিয়ে calculate
Order.objects.annotate(
    total=F('price') * F('quantity')
).filter(total__gt=1000)
```

## Q Objects — Complex Filter

```python
from django.db.models import Q

Post.objects.filter(Q(status='published') | Q(featured=True))   # OR
Post.objects.filter(~Q(status='draft'))                          # NOT
Post.objects.filter(
    Q(status='published') & (Q(author=user) | Q(featured=True)) # complex
)

# Dynamic build
filters = Q()
if search: filters &= Q(title__icontains=search)
if category: filters &= Q(category=category)
posts = Post.objects.filter(filters)
```

## Values ও Values List

```python
# dict হিসেবে — model instance না, fast
Post.objects.values('id', 'title', 'status')
# → [{'id': 1, 'title': 'Hello', ...}, ...]

# tuple হিসেবে
Post.objects.values_list('id', 'title')

# flat list
Post.objects.values_list('id', flat=True)  # → [1, 2, 3]
```

Post.objects.order_by('-created_at', 'title')

Post.objects.filter(...).distinct()

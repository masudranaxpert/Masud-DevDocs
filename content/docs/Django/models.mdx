## Basic Structure

```python
from django.db import models
from django.utils.translation import gettext_lazy as _

class Product(models.Model):
    name        = models.CharField(max_length=200)
    slug        = models.SlugField(unique=True)
    description = models.TextField(blank=True)
    price       = models.DecimalField(max_digits=10, decimal_places=2)
    is_active   = models.BooleanField(default=True)
    created_at  = models.DateTimeField(auto_now_add=True)
    updated_at  = models.DateTimeField(auto_now=True)

    class Meta:
        ordering = ['-created_at']
        verbose_name = _('Product')
        verbose_name_plural = _('Products')
        indexes = [
            models.Index(fields=['slug']),
            models.Index(fields=['is_active', 'created_at']),
        ]

    def __str__(self):
        return self.name

    def get_absolute_url(self):
        from django.urls import reverse
        return reverse('products:detail', kwargs={'slug': self.slug})
```

## blank vs null — সবচেয়ে বেশি ভুল

| | `null=True` | `blank=True` |
|---|---|---|
| কী করে | Database NULL store | Form validation skip |
| CharField তে | ❌ দরকার নেই | ✅ optional হলে |
| DateField/FK তে | ✅ optional হলে | ✅ optional হলে |

```python
# ❌ ভুল — CharField এ null=True দিলে দুইটা empty state: "" এবং NULL
name = models.CharField(max_length=100, null=True, blank=True)

# ✅ সঠিক
name = models.CharField(max_length=100, blank=True, default='')

# ✅ সঠিক — DateField, FK তে null=True ঠিক আছে
published_at = models.DateTimeField(null=True, blank=True)
category = models.ForeignKey(Category, null=True, blank=True, on_delete=models.SET_NULL)
```

## Choices — TextChoices

```python
# ❌ পুরনো — magic strings ছড়িয়ে পড়ে
STATUS_CHOICES = [('draft', 'Draft'), ('published', 'Published')]

# ✅ TextChoices (Django 3.0+)
class Post(models.Model):
    class Status(models.TextChoices):
        DRAFT     = 'draft',     'Draft'
        PUBLISHED = 'published', 'Published'
        ARCHIVED  = 'archived',  'Archived'

    status = models.CharField(
        max_length=20, choices=Status.choices, default=Status.DRAFT,
    )

# ব্যবহার
post.status = Post.Status.PUBLISHED
post.get_status_display()     # → 'Published'
Post.objects.filter(status=Post.Status.PUBLISHED)
```

## Abstract Base Model

```python
class TimeStampedModel(models.Model):
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        abstract = True   # ← এটা না দিলে আলাদা table হবে!

class Product(TimeStampedModel):
    name = models.CharField(max_length=200)
    # created_at, updated_at automatically আছে
```

## Custom Manager

```python
class PublishedManager(models.Manager):
    def get_queryset(self):
        return super().get_queryset().filter(status='published')

class Post(models.Model):
    objects   = models.Manager()      # সব posts
    published = PublishedManager()    # শুধু published

Post.published.all()
Post.published.filter(author=user)
```

## @property vs annotate

```python
class Order(models.Model):
    price    = models.DecimalField(max_digits=10, decimal_places=2)
    quantity = models.PositiveIntegerField()

    @property
    def total(self):
        return self.price * self.quantity

# ❌ filter করা যায় না
Order.objects.filter(total__gt=1000)   # error!

# ✅ annotate দিয়ে filter করো
Order.objects.annotate(total=F('price') * F('quantity')).filter(total__gt=1000)
```

## Index কোথায় দেবে

```python
class Meta:
    indexes = [
        models.Index(fields=['slug']),                      # unique lookup
        models.Index(fields=['email']),                     # frequent filter
        models.Index(fields=['status', 'created_at']),      # composite filter
        models.Index(fields=['-created_at']),               # frequent ORDER BY
    ]
```

`unique=True` field এ automatically index আছে। বেশি index মানে write slow — দরকার মতো দাও।

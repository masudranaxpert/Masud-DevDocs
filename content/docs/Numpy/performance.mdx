## View vs Copy — সবচেয়ে গুরুত্বপূর্ণ

```python
import numpy as np

original = np.array([1, 2, 3, 4, 5])

# --- VIEW --- same memory share করে
view_slice   = original[1:4]      # slice → view
view_reshape = original.reshape(5, 1)  # reshape → usually view
view_T       = original.T         # transpose → view
view_ravel   = original.ravel()   # ravel → usually view

view_slice[0] = 99
print(original)    # → [1 99 3 4 5]  ← original পরিবর্তন হয়েছে!

# --- COPY --- নতুন memory allocate করে
copy_fancy  = original[[0, 1, 2]]  # fancy indexing → copy
copy_bool   = original[original > 2]  # boolean → copy
copy_manual = original.copy()      # explicit copy
copy_astype = original.astype(np.float32)  # astype → copy

copy_fancy[0] = 99
print(original)    # → unchanged

# কোনটা view কিনা check করো
view_slice.base is original   # → True (view)
copy_fancy.base is None       # → True (copy)
```

<Callout type="warn" title="View না Copy — জানা জরুরি">
  View modify করলে original ও পরিবর্তন হয়। Unintended mutation এড়াতে `.copy()` করো।
  কিন্তু unnecessary copy করলে memory ও time দুটোই বেশি লাগে।
</Callout>

## Memory Layout — C vs Fortran Order

```python
# Row-major (C order) — row অনুযায়ী memory তে রাখা (default)
arr_c = np.array([[1, 2, 3], [4, 5, 6]], order='C')
# Memory: [1, 2, 3, 4, 5, 6]

# Column-major (Fortran order) — column অনুযায়ী
arr_f = np.array([[1, 2, 3], [4, 5, 6]], order='F')
# Memory: [1, 4, 2, 5, 3, 6]

# Check করো
arr_c.flags['C_CONTIGUOUS']   # → True
arr_f.flags['F_CONTIGUOUS']   # → True
arr_c.strides                  # → (24, 8) bytes per step
arr_f.strides                  # → (8, 16) bytes per step

# Performance rule:
# Row operation → C order array fast
# Column operation → F order array fast

import time
big = np.random.rand(1000, 1000)

# Row sum — C order fast
start = time.perf_counter()
big.sum(axis=1)
print(f"Row sum: {time.perf_counter()-start:.4f}s")

# Column sum — slightly slower for C order
start = time.perf_counter()
big.sum(axis=0)
print(f"Col sum: {time.perf_counter()-start:.4f}s")
```

## Vectorization — Loop এড়াও

```python
import time
import numpy as np

n = 1_000_000

# --- SLOW: Python loop ---
data = list(range(n))
start = time.perf_counter()
result = [x ** 2 for x in data]
loop_time = time.perf_counter() - start

# --- FAST: NumPy vectorized ---
data_np = np.arange(n)
start = time.perf_counter()
result_np = data_np ** 2
numpy_time = time.perf_counter() - start

print(f"Loop:  {loop_time:.3f}s")
print(f"NumPy: {numpy_time:.4f}s")
print(f"Speedup: {loop_time/numpy_time:.0f}x")
# Typical: 50-100x speedup!

# Conditional loop → np.where
start = time.perf_counter()
result = []
for x in data:
    result.append(x**2 if x % 2 == 0 else x**3)
loop_time = time.perf_counter() - start

start = time.perf_counter()
result = np.where(data_np % 2 == 0, data_np**2, data_np**3)
numpy_time = time.perf_counter() - start
print(f"np.where speedup: {loop_time/numpy_time:.0f}x")
```

## In-place Operations — Memory Save করো

```python
# ❌ নতুন array তৈরি করে — memory বেশি লাগে
a = np.ones((1000, 1000))
a = a + 1          # নতুন array allocate
a = a * 2          # আরেকটা নতুন array

# ✅ In-place — একই memory তে modify
a = np.ones((1000, 1000))
a += 1             # in-place
a *= 2             # in-place

# ufunc out parameter
result = np.empty_like(a)
np.add(a, 1, out=result)      # result তে লিখবে, নতুন allocate না
np.multiply(a, 2, out=a)      # a ই modify হবে
```

images = np.zeros((1000, 224, 224, 3))         # float64 = 1.2 GB

## Profiling — কোথায় slow?

```python
import numpy as np
import time

# Simple timer
def timer(func, *args, n=100):
    start = time.perf_counter()
    for _ in range(n):
        func(*args)
    elapsed = time.perf_counter() - start
    return elapsed / n

arr = np.random.rand(10000)

# Compare approaches
t1 = timer(lambda: np.sum(arr**2))
t2 = timer(lambda: np.dot(arr, arr))   # faster — single pass
print(f"sum(arr**2): {t1*1e6:.2f} μs")
print(f"dot(arr,arr): {t2*1e6:.2f} μs")

# %timeit (Jupyter এ)
# %timeit np.sum(arr**2)
# %timeit np.dot(arr, arr)
```

## Common Performance Mistakes

```python
# ❌ ভুল 1 — অপ্রয়োজনীয় copy
def bad_normalize(arr):
    arr = arr.copy()      # প্রয়োজন নেই যদি modify না করো
    return (arr - arr.mean()) / arr.std()

# ✅ ভালো
def good_normalize(arr):
    return (arr - arr.mean()) / arr.std()

# ❌ ভুল 2 — loop এ append
result = []
for x in range(10000):
    result.append(np.sqrt(x))
result = np.array(result)   # শেষে convert

# ✅ vectorized
result = np.sqrt(np.arange(10000))

# ❌ ভুল 3 — repeated computation
m = np.random.rand(1000, 1000)
for i in range(100):
    result = m.T @ m   # প্রতিবার compute করে

# ✅ precompute
mtm = m.T @ m
for i in range(100):
    # mtm use করো

# ❌ ভুল 4 — unnecessary type conversion
a = np.array([1, 2, 3], dtype=np.int32)
b = np.array([1.0, 2.0, 3.0], dtype=np.float64)
result = a + b   # a কে float64 এ convert করে — implicit upcasting

# ✅ explicit dtype
a = np.array([1, 2, 3], dtype=np.float32)
b = np.array([1.0, 2.0, 3.0], dtype=np.float32)
result = a + b   # no conversion needed
```

## einsum — Einstein Summation (Advanced)

```python
# np.einsum — complex array operations clearly express করো
a = np.random.rand(3, 4)
b = np.random.rand(4, 5)

# Matrix multiplication
np.einsum('ij,jk->ik', a, b)    # → (3, 5), same as a @ b

# Trace (diagonal sum)
m = np.random.rand(4, 4)
np.einsum('ii', m)               # → trace, same as np.trace(m)

# Batch dot product
a_batch = np.random.rand(10, 3)  # 10 vectors of length 3
b_batch = np.random.rand(10, 3)
np.einsum('ij,ij->i', a_batch, b_batch)  # → 10 dot products

# Outer product
a = np.array([1, 2, 3])
b = np.array([4, 5, 6])
np.einsum('i,j->ij', a, b)       # → (3, 3) outer product
```
